%%Use CPLEX in the Distribution System\nclear,clc\nclose all;\npath_base = mfilename('fullpath');i=findstr(path_base,'\');path_base=path_base(1:i(end));\n\n\n%% Standard parameters for initial conversion examples of grid structure (integrated with matpower, using standard case file input)\n%--------------Testing Case Selection (IEEE33)-----------------\ncase_name_or=case33bw; % Standard case name\ncase_name=Transfer_node_num_to_consecutive(case_name_or);% Change the bus name to consecutive numbers Modify the node number in matpower to consecutive numbers.\n[ref,pv, pq] = bustypes(case_name.bus, case_name.gen); % Output the number of balanced nodes, PV nodes, and PQ nodes\nBus_Num=size(case_name.bus,1); % Number of nodes\nline_active=find(case_name.branch(:,11)==1); % Power Lines\nBranch_Num=size(line_active,1);\n%% Initialize the parameters for this example\nPV_Rate=0.7; %The power output of photovoltaic (PV) is the power output per unit capacity, generally less than 0.9.\n[Device_Info] = Init_Device_Info(case_name,1);% Initialize device parameters\n[Pos_In_OutPut] = Define_Pos_VariableVector(case_name,Device_Info,Bus_Num,Branch_Num);\n%---------------Standard Case  Standard Example (currently selected for overvoltage))--------------------------\n[Input_Standard] = Generate_Standard_Case_Vector(case_name,Device_Info,PV_Rate);\n[Output_NR,Input_NR,casename_NR,results_NR]=NR_PF_Cal(case_name,Input_Standard,Device_Info);\n figure;plot(results_NR.bus(:,8));title('Original Case');xlabel('Node number');ylabel('Node voltagep.u.')\n % The above is for reading and obtaining the current path, no need to modify.\n \n%%\n%--------------Init parameter-----------------------------------\nConstraint.Voltage.max=1.05;    % Voltage Limit\nConstraint.Voltage.min=0.95;    %Voltage Lower Limit\n\n\nConstraint.PV_Rate=PV_Rate;  % Power ratio of photovoltaic output\n\n\nConstraint.PVinverter_P=Input_Standard(Pos_In_OutPut.Pos_P_in_Inventer,1)*1; %MW % Distributed power (Photovoltaic inverter) active power\nConstraint.PVinverter_Q=Input_Standard(Pos_In_OutPut.Pos_Q_in_Inventer,1)*1; %MVar % Distributed Power Source (Photovoltaic Inverter) Reactive Power\nConstraint.P_Load=[0;Input_Standard(Pos_In_OutPut.Pos_P_Load,1)]*1; %Node Load Active Power MW\nConstraint.Q_Load=[0;Input_Standard(Pos_In_OutPut.Pos_Q_Load,1)]*1; % Node Load Reactive Power MVar\nConstraint.INclude_PV_node=Device_Info.INclude_PV_node; % Node numbers containing distributed power sources (photovoltaic inverters)\nConstraint.INclude_PV_S=Device_Info.INclude_PV_S';% Installed capacity of distributed power sources (photovoltaic inverters) MW\n\n\n%% CPLEX Optimization\n    %  [Distflow][SOCP] Power Flow \n        [Result_CPLEX] = CPLEX_DLPF_Optimization(casename_NR,Device_Info.INclude_PV_node,Constraint);\n        % objective: Minimize the power loss\n        % Optimization goal: Minimize network losses\n        % constraint:Distflow power flow constraints; PV inverter operation constraints; SOCP constraints\n        % Constraints: DLPF power flow constraints; photovoltaic inverter operation constraints; second-order cone relaxation\n%% Draw Figure\n\n\n%         figure;\n%         subplot(2,3,1);plot(Result_CPLEX);hold on;plot(results_NR.bus(:,8));  %.v_m/case_name..bus(ref,10))\n%         xlabel('Node number');ylabel('Node voltage/p.u.');legend('Optimized','Optimized before');\n%         subplot(2,3,2);plot(Result_CPLEX.Pline);hold on;plot(results_NR.branch(line_active,14));\n% xlabel('Branch number');ylabel('Branch active power/MW');legend('Optimized','Before optimization');\n%         subplot(2,3,3);plot(Result_CPLEX.Qline);hold on;plot(results_NR.branch(line_active,15));\n% xlabel('Branch number');ylabel('Reactive power/MVar of branches');legend('Optimized','Before optimization');\n%         subplot(2,3,4);bar( Device_Info.INclude_PV_node,[Result_CPLEX.PV_P,Input_Standard(Pos_In_OutPut.Pos_P_in_Inventer,:)]);\n%         xlabel('Access to photovoltaic node number');ylabel('Photovoltaic output active power/MW');legend('Optimized','Optimized before');\n%         subplot(2,3,5);bar( Device_Info.INclude_PV_node,[Result_CPLEX.PV_Q,Input_Standard(Pos_In_OutPut.Pos_Q_in_Inventer,:)]);\n%         xlabel('Access to photovoltaic node number');ylabel('Photovoltaic output reactive power/MVar');legend('Optimized','Optimized before');\n%         set(gcf,'unit','centimeters','position',[1,2,25,15]);\n% f MW\n',sum(abs(results_NR.branch(line_active,14)+results_NR.branch(line_active,16))));\n% fprintf('Optimized network loss: %f MW', Result_CPLEX.objective);\n\n\n\n\n\n\n
